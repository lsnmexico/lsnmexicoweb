// CONFIGURATION - SZERVER BEÁLLÍTÁSOK const SERVER_CONFIG = { serverIp: "94.156.37.78", serverPort: "30120", updateInterval: 5000 }; const statusElem = document.getElementById('server-status'); const countElem = document.getElementById('player-count'); const maxElem = document.getElementById('max-players'); // Valós idejű státusz lekérdezés - Direct Query async function fetchServerStatus() { try { console.log("🔍 Szerver státusz lekérdezésének indítása..."); // Direct FiveM szerver query - /info.json const infoUrl = `http://${SERVER_CONFIG.serverIp}:${SERVER_CONFIG.serverPort}/info.json`; console.log("📡 Lekérdezés:", infoUrl); const response = await fetch(infoUrl, { method: 'GET', mode: 'cors', cache: 'no-cache' }); console.log("Válasz státusza:", response.status); if (response.ok) { const data = await response.json(); console.log("✅ Szerver adatok:", data); const currentPlayers = data.clients || 0; const maxPlayers = data.sv_maxclients || 10; // Update DOM countElem.textContent = currentPlayers; maxElem.textContent = maxPlayers; statusElem.textContent = "Online"; statusElem.className = "inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[10px] font-bold bg-green-500/10 text-green-400 uppercase tracking-widest border border-green-500/20"; console.log(`✅ Szerver Online: ${currentPlayers}/${maxPlayers} játékos`); } else { console.error("❌ HTTP hiba:", response.status); setOfflineStatus(); } } catch (error) { console.error("❌ Lekérdezési hiba:", error.message); setOfflineStatus(); } } function setOfflineStatus() { statusElem.textContent = "Offline"; statusElem.className = "inline-flex items-center gap-1.5 px-2.5 py-1 rounded-full text-[10px] font-bold bg-red-500/10 text-red-400 uppercase tracking-widest border border-red-500/20"; countElem.textContent = "0"; maxElem.textContent = "0"; console.warn("⚠️ Szerver jelenleg offline"); } // Első futtatás azonnal fetchServerStatus(); // Periodikus frissítés beállítása setInterval(fetchServerStatus, SERVER_CONFIG.updateInterval);
